\begin{oefening}[Rekenvolgorde]{\oefB\ESS}
    \\
    
    Bereken exact. Let goed op de volgorde van de bewerkingen.
    
    \begin{multicols}{2}
    
    \begin{question}
    $18-3(2+4)=\answer{0}$
    
    \begin{oplossing}
    \[
    18-3(2+4)
    =18-3\cdot6
    =18-18
    =0.
    \]
    \end{oplossing}
    \end{question}
    
    \begin{question}
    $5+2^3\cdot3-7=\answer{22}$
    
    \begin{oplossing}
    \[
    5+2^3\cdot3-7
    =5+8\cdot3-7
    =5+24-7
    =22.
    \]
    \end{oplossing}
    \end{question}
    
    \begin{question}
    $4(7-3)-2(5+1)=\answer{4}$
    
    \begin{oplossing}
    \[
    4(7-3)-2(5+1)
    =4\cdot4-2\cdot6
    =16-12
    =4.
    \]
    \end{oplossing}
    \end{question}
    
    \begin{question}
    $\dfrac{24}{3\cdot2}+5=\answer{9}$
    
    \begin{oplossing}
    \[
    \dfrac{24}{3\cdot2}+5
    =\dfrac{24}{6}+5
    =4+5
    =9.
    \]
    \end{oplossing}
    \end{question}
    
    \begin{question}
    $3-[2-4(1-3)]=\answer{-7}$
    
    \begin{oplossing}
    \[
    3-[2-4(1-3)]
    =3-[2-4(-2)]
    =3-[2+8]
    =3-10
    =-7.
    \]
    \end{oplossing}
    \end{question}
    
    \begin{question}
    $\dfrac{2^4-4}{3}+1=\answer{5}$
    
    \begin{oplossing}
    \[
    \dfrac{2^4-4}{3}+1
    =\dfrac{16-4}{3}+1
    =\dfrac{12}{3}+1
    =5.
    \]
    \end{oplossing}
    \end{question}
    
    \begin{question}
    $7+3(5-2)^2=\answer{34}$
    
    \begin{oplossing}
    \[
    7+3(5-2)^2
    =7+3\cdot3^2
    =7+27
    =34.
    \]
    \end{oplossing}
    \end{question}
    
    \begin{question}
    $\dfrac{48}{2(3+1)}=\answer{6}$
    
    \begin{oplossing}
    \[
    \dfrac{48}{2(3+1)}
    =\dfrac{48}{8}
    =6.
    \]
    \end{oplossing}
    \end{question}
    
    \begin{question}
    $6-2[3-(5-8)]=\answer{-6}$
    
    \begin{oplossing}
    \[
    6-2[3-(5-8)]
    =6-2[3-(-3)]
    =6-12
    =-6.
    \]
    \end{oplossing}
    \end{question}
    
    \begin{question}
    $\dfrac{3^2+7}{2^2}=\answer{4}$
    
    \begin{oplossing}
    \[
    \dfrac{3^2+7}{2^2}
    =\dfrac{9+7}{4}
    =4.
    \]
    \end{oplossing}
    \end{question}
    
    \begin{question}
    $20-2^2(3+1)+5=\answer{9}$
    
    \begin{oplossing}
    \[
    20-2^2(3+1)+5
    =20-4\cdot4+5
    =9.
    \]
    \end{oplossing}
    \end{question}
    
    \begin{question}
    $2\left[5-3(2-4)\right]-6=\answer{16}$
    
    \begin{oplossing}
    \[
    2\left[5-3(2-4)\right]-6
    =2[5-3(-2)]-6
    =2\cdot11-6
    =16.
    \]
    \end{oplossing}
    \end{question}
    
    \end{multicols}
    
    \end{oefening}

    \begin{oefening}[Breuken]{\oefB\ESS}
        \\
        
        Werk uit en vereenvoudig.
        
        \begin{multicols}{2}
        
        \begin{question}
        $\dfrac34+\dfrac56=\answer{\dfrac{19}{12}}$
        
        \begin{oplossing}
        \[
        \dfrac34+\dfrac56
        =\dfrac9{12}+\dfrac{10}{12}
        =\dfrac{19}{12}.
        \]
        \end{oplossing}
        \end{question}
        
        
        \begin{question}
        $\dfrac78-\dfrac13=\answer{\dfrac{13}{24}}$
        
        \begin{oplossing}
        \[
        \dfrac78-\dfrac13
        =\dfrac{21}{24}-\dfrac8{24}
        =\dfrac{13}{24}.
        \]
        \end{oplossing}
        \end{question}
        
        
        \begin{question}
        $\dfrac5{12}+\dfrac7{18}=\answer{\dfrac{29}{36}}$
        
        \begin{oplossing}
        \[
        \dfrac5{12}+\dfrac7{18}
        =\dfrac{15}{36}+\dfrac{14}{36}
        =\dfrac{29}{36}.
        \]
        \end{oplossing}
        \end{question}
        
        
        \begin{question}
        $\dfrac{11}{15}-\dfrac7{20}=\answer{\dfrac{23}{60}}$
        
        \begin{oplossing}
        \[
        \dfrac{11}{15}-\dfrac7{20}
        =\dfrac{44}{60}-\dfrac{21}{60}
        =\dfrac{23}{60}.
        \]
        \end{oplossing}
        \end{question}
        
        
        \begin{question}
        $\dfrac23\cdot\dfrac9{10}=\answer{\dfrac35}$
        
        \begin{oplossing}
        \[
        \dfrac23\cdot\dfrac9{10}
        =\dfrac{18}{30}
        =\dfrac35.
        \]
        \end{oplossing}
        \end{question}
        
        
        \begin{question}
        $\dfrac56:\dfrac{10}{9}=\answer{\dfrac34}$
        
        \begin{oplossing}
        Delen door een breuk betekent vermenigvuldigen met het omgekeerde:
        \[
        \dfrac56:\dfrac{10}{9}
        =\dfrac56\cdot\dfrac9{10}
        =\dfrac{45}{60}
        =\dfrac34.
        \]
        \end{oplossing}
        \end{question}
        
        
        \begin{question}
        $\dfrac{14}{15}\cdot\dfrac{25}{21}
        =\answer{\dfrac{10}{9}}$
        
        \begin{oplossing}
        We vereenvoudigen vóór het vermenigvuldigen:
        \[
        \dfrac{14}{15}\cdot\dfrac{25}{21}
        =\dfrac{2}{3}\cdot\dfrac{5}{3}
        =\dfrac{10}{9}.
        \]
        \end{oplossing}
        \end{question}
        
        
        \begin{question}
        $\dfrac{7}{12}:\dfrac{14}{9}
        =\answer{\dfrac38}$
        
        \begin{oplossing}
        \[
        \dfrac{7}{12}:\dfrac{14}{9}
        =\dfrac{7}{12}\cdot\dfrac9{14}
        =\dfrac{63}{168}
        =\dfrac38.
        \]
        \end{oplossing}
        \end{question}
        
        
        \begin{question}
        $\dfrac{3x}{5}+\dfrac{2x}{3}
        =\answer{\dfrac{19x}{15}}$
        
        \begin{oplossing}
        \[
        \dfrac{3x}{5}+\dfrac{2x}{3}
        =\dfrac{9x}{15}+\dfrac{10x}{15}
        =\dfrac{19x}{15}.
        \]
        \end{oplossing}
        \end{question}
        
        
        \begin{question}
        $\dfrac{5a}{8}-\dfrac{a}{6}
        =\answer{\dfrac{11a}{24}}$
        
        \begin{oplossing}
        \[
        \dfrac{5a}{8}-\dfrac{a}{6}
        =\dfrac{15a}{24}-\dfrac{4a}{24}
        =\dfrac{11a}{24}.
        \]
        \end{oplossing}
        \end{question}
        
        
        \begin{question}
        $\dfrac{4x}{9}\cdot\dfrac{3}{2x^2}
        =\answer{\dfrac{2}{3x}}$
        
        \begin{oplossing}
        Omdat $x$ in een noemer voorkomt, geldt $x\neq0$.
        \[
        \dfrac{4x}{9}\cdot\dfrac{3}{2x^2}
        =\dfrac{12x}{18x^2}
        =\dfrac{2}{3x}.
        \]
        \end{oplossing}
        \end{question}
        
        
        \begin{question}
        $\dfrac{6x^2}{5}\cdot\dfrac{10}{9x}
        =\answer{\dfrac{4x}{3}}$
        
        \begin{oplossing}
        Voor $x\neq0$:
        \[
        \dfrac{6x^2}{5}\cdot\dfrac{10}{9x}
        =\dfrac{60x^2}{45x}
        =\dfrac{4x}{3}.
        \]
        \end{oplossing}
        \end{question}
        
        
        \begin{question}
        $\dfrac{x^2-4}{x^2+2x}
        =\answer{\dfrac{x-2}{x}}$
        
        \begin{oplossing}
        De oorspronkelijke noemer is $x(x+2)$, dus
        \[
        x\neq0
        \qquad\text{en}\qquad
        x\neq-2.
        \]
        Verder:
        \[
        \dfrac{x^2-4}{x^2+2x}
        =\dfrac{(x-2)(x+2)}{x(x+2)}
        =\dfrac{x-2}{x}.
        \]
        \end{oplossing}
        \end{question}
        
        
        \begin{question}
        $\dfrac{x^2-9}{x^2-6x+9}
        =\answer{\dfrac{x+3}{x-3}}$
        
        \begin{oplossing}
        De oorspronkelijke noemer is $(x-3)^2$, dus
        \[
        x\neq3.
        \]
        Verder:
        \[
        \dfrac{x^2-9}{x^2-6x+9}
        =\dfrac{(x-3)(x+3)}{(x-3)^2}
        =\dfrac{x+3}{x-3}.
        \]
        \end{oplossing}
        \end{question}
        
        
        \begin{question}
        $\dfrac{x^2-16}{x^2-4x}
        =\answer{\dfrac{x+4}{x}}$
        
        \begin{oplossing}
        De oorspronkelijke noemer is $x(x-4)$, dus
        \[
        x\neq0
        \qquad\text{en}\qquad
        x\neq4.
        \]
        Verder:
        \[
        \dfrac{x^2-16}{x^2-4x}
        =\dfrac{(x-4)(x+4)}{x(x-4)}
        =\dfrac{x+4}{x}.
        \]
        \end{oplossing}
        \end{question}
        
        
        \begin{question}
        $\dfrac{2x^2+6x}{4x}
        =\answer{\dfrac{x+3}{2}}$
        
        \begin{oplossing}
        Voor $x\neq0$:
        \[
        \dfrac{2x^2+6x}{4x}
        =\dfrac{2x(x+3)}{4x}
        =\dfrac{x+3}{2}.
        \]
        \end{oplossing}
        \end{question}
        
        \end{multicols}
        
        \end{oefening}

        \begin{oefening}[Gemengde breuken]{\oefB}
            \\
            
            Werk zo ver mogelijk uit.
            
            \begin{multicols}{2}
            
            \begin{question}
            $\dfrac1x+\dfrac1{x+1}
            =\answer{\dfrac{2x+1}{x(x+1)}}$
            
            \begin{oplossing}
            Met $x\neq0,-1$:
            \[
            \dfrac1x+\dfrac1{x+1}
            =\dfrac{x+1+x}{x(x+1)}
            =\dfrac{2x+1}{x(x+1)}.
            \]
            \end{oplossing}
            \end{question}
            
            
            \begin{question}
            $\dfrac{x}{x+2}+\dfrac3{x+2}
            =\answer{\dfrac{x+3}{x+2}}$
            
            \begin{oplossing}
            Voor $x\neq-2$:
            \[
            \dfrac{x}{x+2}+\dfrac3{x+2}
            =\dfrac{x+3}{x+2}.
            \]
            \end{oplossing}
            \end{question}
            
            
            \begin{question}
            $\dfrac2{x-1}-\dfrac5{x+1}
            =\answer{\dfrac{-3x+7}{x^2-1}}$
            
            \begin{oplossing}
            Voor $x\neq1,-1$:
            \[
            \dfrac2{x-1}-\dfrac5{x+1}
            =\dfrac{2(x+1)-5(x-1)}{(x-1)(x+1)}
            \]
            \[
            =\dfrac{2x+2-5x+5}{x^2-1}
            =\dfrac{-3x+7}{x^2-1}.
            \]
            \end{oplossing}
            \end{question}
            
            
            \begin{question}
            $\dfrac3x+\dfrac2{x-2}
            =\answer{\dfrac{5x-6}{x(x-2)}}$
            
            \begin{oplossing}
            Voor $x\neq0,2$:
            \[
            \dfrac3x+\dfrac2{x-2}
            =\dfrac{3(x-2)+2x}{x(x-2)}
            \]
            \[
            =\dfrac{3x-6+2x}{x(x-2)}
            =\dfrac{5x-6}{x(x-2)}.
            \]
            \end{oplossing}
            \end{question}
            
            
            \begin{question}
            $\dfrac4{x+1}-\dfrac1{x-1}
            =\answer{\dfrac{3x-5}{x^2-1}}$
            
            \begin{oplossing}
            Voor $x\neq-1,1$:
            \[
            \dfrac4{x+1}-\dfrac1{x-1}
            =\dfrac{4(x-1)-(x+1)}{(x+1)(x-1)}
            \]
            \[
            =\dfrac{4x-4-x-1}{x^2-1}
            =\dfrac{3x-5}{x^2-1}.
            \]
            \end{oplossing}
            \end{question}
            
            
            \begin{question}
            $\dfrac{x^2+5x+6}{x+2}
            =\answer{x+3}$
            
            \begin{oplossing}
            Voor $x\neq-2$:
            \[
            \dfrac{x^2+5x+6}{x+2}
            =\dfrac{(x+2)(x+3)}{x+2}
            =x+3.
            \]
            \end{oplossing}
            \end{question}
            
            
            \begin{question}
            $\dfrac{2x^2-8}{x+2}
            =\answer{2x-4}$
            
            \begin{oplossing}
            Voor $x\neq-2$:
            \[
            \dfrac{2x^2-8}{x+2}
            =\dfrac{2(x^2-4)}{x+2}
            \]
            \[
            =\dfrac{2(x-2)(x+2)}{x+2}
            =2(x-2)
            =2x-4.
            \]
            \end{oplossing}
            \end{question}
            
            
            \begin{question}
            $\dfrac{x^2-x-6}{x-3}
            =\answer{x+2}$
            
            \begin{oplossing}
            Voor $x\neq3$:
            \[
            \dfrac{x^2-x-6}{x-3}
            =\dfrac{(x-3)(x+2)}{x-3}
            =x+2.
            \]
            \end{oplossing}
            \end{question}
            
            
            \begin{question}
            $\dfrac{x^2-9}{x+3}
            =\answer{x-3}$
            
            \begin{oplossing}
            Voor $x\neq-3$:
            \[
            \dfrac{x^2-9}{x+3}
            =\dfrac{(x-3)(x+3)}{x+3}
            =x-3.
            \]
            \end{oplossing}
            \end{question}
            
            
            \begin{question}
            $\dfrac{3x^2+6x}{6x}
            =\answer{\dfrac{x+2}{2}}$
            
            \begin{oplossing}
            Voor $x\neq0$:
            \[
            \dfrac{3x^2+6x}{6x}
            =\dfrac{3x(x+2)}{6x}
            =\dfrac{x+2}{2}.
            \]
            \end{oplossing}
            \end{question}
            
            
            \begin{question}
            $\dfrac{\frac{12}{5}}{3}
            =\answer{\dfrac45}$
            
            \begin{oplossing}
            \[
            \dfrac{\frac{12}{5}}{3}
            =\dfrac{12}{5}\cdot\dfrac13
            =\dfrac45.
            \]
            \end{oplossing}
            \end{question}
            
            
            \begin{question}
            $\dfrac{\frac34}{\frac56}
            =\answer{\dfrac9{10}}$
            
            \begin{oplossing}
            \[
            \dfrac{\frac34}{\frac56}
            =\dfrac34:\dfrac56
            =\dfrac34\cdot\dfrac65
            =\dfrac{18}{20}
            =\dfrac9{10}.
            \]
            \end{oplossing}
            \end{question}
            
            
            \begin{question}
            $\dfrac{\frac{2x}{3}}{\frac{x}{4}}
            =\answer{\dfrac83}$
            
            \begin{oplossing}
            Voor $x\neq0$:
            \[
            \dfrac{\frac{2x}{3}}{\frac{x}{4}}
            =\dfrac{2x}{3}\cdot\dfrac4x

            \]
            \end{oplossing}
            \end{question}
            
            
            \begin{question}
            $\dfrac{\frac{x^2-4}{x}}{x+2}
            =\answer{\dfrac{x-2}{x}}$

            \[
            \dfrac{\frac{x^2-4}{x}}{x+2}
            =\dfrac{x^2-4}{x(x+2)}
            \]
            \[
            =\dfrac{(x-2)(x+2)}{x(x+2)}

            \]
            \end{oplossing}
            \end{question}
            
            
            \begin{question}
            $\dfrac{\frac{x^2-1}{x}}{x-1}
            =\answer{\dfrac{x+1}{x}}$
            
            \begin{oplossing}

            \[
            \dfrac{\frac{x^2-1}{x}}{x-1}
            =\dfrac{x^2-1}{x(x-1)}
            \]
            \[
            =\dfrac{(x-1)(x+1)}{x(x-1)}
            =\dfrac{x+1}{x}.
            \]

            \end{question}
            
            
            \begin{question}
            $\dfrac{\frac{2x^2+4x}{3x}}{x+2}
            =\answer{\dfrac23}$

            \]

            \end{multicols}

            \begin{oefening}[Slim rekenen en schatten]{\oefU\ESS}
                \\
                
                Geef eerst zonder rekenmachine een redelijke schatting. Bereken daarna nauwkeurig.
                
                \begin{multicols}{2}
                
                \begin{question}
                $\dfrac{198\cdot4{,}9}{10{,}2}\approx\answer{95.12}$
                
                \begin{oplossing}
                We schatten eerst:
                \[
                \dfrac{200\cdot5}{10}\approx100.
                \]
                
                Nauwkeuriger:
                \[
                \dfrac{198\cdot4{,}9}{10{,}2}
                =\dfrac{970{,}2}{10{,}2}
                \approx95{,}12.
                \]
                
                De uitkomst ligt inderdaad in de buurt van onze schatting.
                \end{oplossing}
                \end{question}
                
                
                \begin{question}
                $19{,}8^2=\answer{392.04}$
                
                \begin{oplossing}
                Omdat $19{,}8\approx20$, verwachten we ongeveer
                \[
                20^2=400.
                \]
                
                Nauwkeurig:
                \[
                19{,}8^2
                =(20-0{,}2)^2
                =400-8+0{,}04
                =392{,}04.
                \]
                \end{oplossing}
                \end{question}
                
                
                \begin{question}
                $\dfrac{0{,}049\cdot201}{0{,}99}\approx\answer{9.95}$
                
                \begin{oplossing}
                We schatten:
                \[
                \dfrac{0{,}05\cdot200}{1}\approx10.
                \]
                
                Nauwkeuriger:
                \[
                \dfrac{0{,}049\cdot201}{0{,}99}
                =\dfrac{9{,}849}{0{,}99}
                \approx9{,}95.
                \]
                \end{oplossing}
                \end{question}
                
                
                \begin{question}
                $\dfrac{49{,}8\cdot20{,}1}{10{,}2}\approx\answer{98.15}$
                
                \begin{oplossing}
                We schatten:
                \[
                \dfrac{50\cdot20}{10}\approx100.
                \]
                
                Nauwkeuriger:
                \[
                \dfrac{49{,}8\cdot20{,}1}{10{,}2}
                =\dfrac{1000{,}98}{10{,}2}
                \approx98{,}14.
                \]
                
                Dit ligt dicht bij de geschatte waarde $100$.
                \end{oplossing}
                \end{question}
                
                
                \begin{question}
                $5{,}02\cdot19{,}9\approx\answer{99.90}$
                
                \begin{oplossing}
                We schatten:
                \[
                5{,}02\cdot19{,}9
                \approx5\cdot20
                =100.
                \]
                
                Nauwkeuriger:
                \[
                5{,}02\cdot19{,}9
                =99{,}898
                \approx99{,}90.
                \]
                \end{oplossing}
                \end{question}
                
                
                \begin{question}
                $\dfrac{399}{8{,}1}\approx\answer{49.26}$
                
                \begin{oplossing}
                We schatten:
                \[
                \dfrac{400}{8}=50.
                \]
                
                Nauwkeuriger:
                \[
                \dfrac{399}{8{,}1}
                \approx49{,}26.
                \]
                
                De nauwkeurige waarde ligt dus zoals verwacht dicht bij $50$.
                \end{oplossing}
                \end{question}
                
                
                \begin{question}
                $29{,}9^2\approx\answer{894.01}$
                
                \begin{oplossing}
                Omdat $29{,}9\approx30$, verwachten we ongeveer
                \[
                30^2=900.
                \]
                
                Nauwkeurig:
                \[
                29{,}9^2
                =(30-0{,}1)^2
                =900-6+0{,}01
                =894{,}01.
                \]
                \end{oplossing}
                \end{question}
                
                
                \begin{question}
                $\dfrac{1002\cdot0{,}098}{2{,}01}\approx\answer{48.86}$
                
                \begin{oplossing}
                We schatten:
                \[
                \dfrac{1000\cdot0{,}1}{2}
                =50.
                \]
                
                Nauwkeuriger:
                \[
                \dfrac{1002\cdot0{,}098}{2{,}01}
                =\dfrac{98{,}196}{2{,}01}
                \approx48{,}85.
                \]
                
                Dit is goed verenigbaar met onze schatting.
                \end{oplossing}
                \end{question}
                
                
                \begin{question}
                Een leerling vindt voor
                \[
                \dfrac{603}{19{,}8}
                \]
                de uitkomst $3{,}05$.
                
                Leg uit waarom je zonder herberekenen al weet dat dit fout is.
                
                \begin{oplossing}
                We schatten:
                \[
                603\approx600
                \qquad\text{en}\qquad
                19{,}8\approx20.
                \]
                
                Dus
                \[
                \dfrac{603}{19{,}8}
                \approx
                \dfrac{600}{20}
                =30.
                \]
                
                Een uitkomst rond $3$ kan dus niet kloppen. Ze is ongeveer een factor $10$ te klein.
                \end{oplossing}
                \end{question}
                
                
                \begin{question}
                Een leerling berekent
                \[
                49{,}7\cdot20{,}2=10039{,}4.
                \]
                
                Leg zonder de vermenigvuldiging opnieuw uit te voeren uit waarom dit antwoord fout is.
                
                \begin{oplossing}
                We schatten:
                \[
                49{,}7\approx50
                \qquad\text{en}\qquad
                20{,}2\approx20.
                \]
                
                Daarom verwachten we
                \[
                50\cdot20=1000.
                \]
                
                Een antwoord rond $10\,000$ is dus ongeveer een factor $10$ te groot.
                \end{oplossing}
                \end{question}
                
                
                \begin{question}
                Een leerling vindt
                \[
                \sqrt{398}=199.
                \]
                
                Leg met een schatting uit waarom dit onmogelijk is.
                
                \begin{oplossing}
                Omdat
                \[
                398\approx400
                \]
                en
                \[
                \sqrt{400}=20,
                \]
                moet $\sqrt{398}$ iets kleiner zijn dan $20$.
                
                Een uitkomst van $199$ kan dus onmogelijk correct zijn.
                \end{oplossing}
                \end{question}
                
                
                \begin{question}
                Een leerling berekent
                \[
                \dfrac{0{,}51\cdot198}{10{,}1}
                \]
                en krijgt als resultaat $0{,}999$.
                
                Onderzoek met een schatting of dit resultaat aannemelijk is.
                
                \begin{oplossing}
                We schatten:
                \[
                0{,}51\approx0{,}5,
                \qquad
                198\approx200,
                \qquad
                10{,}1\approx10.
                \]
                
                Dus
                \[
                \dfrac{0{,}51\cdot198}{10{,}1}
                \approx
                \dfrac{0{,}5\cdot200}{10}
                =\dfrac{100}{10}
                =10.
                \]
                
                Een resultaat rond $1$ is daarom niet aannemelijk. Het antwoord is ongeveer een factor $10$ te klein.
                \end{oplossing}
                \end{question}
                
                \end{multicols}

            \begin{oefening}[Ordegrootte in de wetenschappen]{\oefU}
                \\
                
                Bij deze oefening bedoelen we met de \emph{ordegrootte} de macht van $10$
                die het dichtst bij de gegeven waarde ligt.
                
                Een bacterie is ongeveer $2\cdot10^{-6}\,\mathrm m$ lang en een menselijke haar
                ongeveer $7\cdot10^{-5}\,\mathrm m$ dik.
                
                \begin{multicols}{2}
                
                \begin{question}
                Hoeveel keer groter is de dikte van de haar dan de lengte van de bacterie?
                
                Antwoord: $\answer{35}$ keer.
                
                \begin{oplossing}
                We vergelijken beide lengtes:
                \[
                \dfrac{7\cdot10^{-5}}{2\cdot10^{-6}}
                =\dfrac72\cdot10^{-5-(-6)}
                \]
                \[
                =3{,}5\cdot10
                =35.
                \]
                
                De haar is dus ongeveer $35$ keer dikker dan de bacterie lang is.
                \end{oplossing}
                \end{question}
                
                
                \begin{question}
                Wat is de ordegrootte van de lengte van de bacterie?
                
                \[
                \answer{10^{-6}}\,\mathrm m
                \]
                
                \begin{oplossing}
                De lengte is
                \[
                2\cdot10^{-6}\,\mathrm m.
                \]
                
                Deze waarde ligt dichter bij
                \[
                10^{-6}\,\mathrm m
                \]
                dan bij $10^{-5}\,\mathrm m$.
                
                De ordegrootte is dus
                \[
                10^{-6}\,\mathrm m.
                \]
                \end{oplossing}
                \end{question}
                
                
                \begin{question}
                Wat is de ordegrootte van de dikte van de haar?
                
                \[
                \answer{10^{-4}}\,\mathrm m
                \]
                
                \begin{oplossing}
                De dikte is
                \[
                7\cdot10^{-5}\,\mathrm m.
                \]
                
                Deze waarde ligt dichter bij
                \[
                10^{-4}\,\mathrm m
                \]
                dan bij $10^{-5}\,\mathrm m$.
                
                De ordegrootte is dus
                \[
                10^{-4}\,\mathrm m.
                \]
                \end{oplossing}
                \end{question}
                
                
                \begin{question}
                Hoeveel ordes van grootte verschillen de bacterie en de haar?
                
                Antwoord: $\answer{2}$ ordes van grootte.
                
                \begin{oplossing}

                \[
                10^{-6}\,\mathrm m
                \qquad\text{en}\qquad
                10^{-4}\,\mathrm m.
                \]
                
                Het verschil tussen de exponenten is
                \[
                -4-(-6)=2.
                \]
                
                Ze verschillen dus twee ordes van grootte.
                \end{oplossing}
                \end{question}
                
                
                \begin{question}
                Een virus heeft een diameter van ongeveer
                \[
                1{,}0\cdot10^{-7}\,\mathrm m.
                \]
                
                Hoeveel keer langer is de bacterie dan de diameter van het virus?
                
                Antwoord: $\answer{20}$ keer.
                
                \begin{oplossing}
                \[
                \dfrac{2\cdot10^{-6}}{1{,}0\cdot10^{-7}}
                =2\cdot10
                =20.
                \]
                
                De bacterie is dus ongeveer $20$ keer langer.
                \end{oplossing}
                \end{question}
                
                
                \begin{question}
                Een menselijke rode bloedcel heeft een diameter van ongeveer
                \[
                8\cdot10^{-6}\,\mathrm m.
                \]
                
                Hoeveel keer groter is die diameter dan de lengte van de bacterie?
                
                Antwoord: $\answer{4}$ keer.
                
                \begin{oplossing}
                \[
                \dfrac{8\cdot10^{-6}}{2\cdot10^{-6}}
                =\dfrac82\cdot10^0
                =4.
                \]
                
                De diameter van de rode bloedcel is dus ongeveer $4$ keer groter.
                \end{oplossing}
                \end{question}
                
                
                \begin{question}
                Een stofdeeltje heeft een diameter van ongeveer
                \[
                3\cdot10^{-4}\,\mathrm m.
                \]
                
                Hoeveel keer groter is dit dan de lengte van de bacterie?
                
                Antwoord: $\answer{150}$ keer.
                
                \begin{oplossing}
                \[
                \dfrac{3\cdot10^{-4}}{2\cdot10^{-6}}
                =\dfrac32\cdot10^2
                \]
                \[
                =1{,}5\cdot100
                =150.
                \]
                \end{oplossing}
                \end{question}
                
                
                \begin{question}
                De afstand van de aarde tot de zon is ongeveer
                \[
                1{,}5\cdot10^{11}\,\mathrm m.
                \]
                
                Wat is de ordegrootte van deze afstand?
                
                \[
                \answer{10^{11}}\,\mathrm m
                \]
                
                \begin{oplossing}
                De waarde
                \[
                1{,}5\cdot10^{11}
                \]
                ligt dichter bij $10^{11}$ dan bij $10^{12}$.
                
                De ordegrootte is dus
                \[
                10^{11}\,\mathrm m.
                \]
                \end{oplossing}
                \end{question}
                
                
                \begin{question}
                Een cel heeft een typische afmeting van
                \[
                1\cdot10^{-5}\,\mathrm m,
                \]
                terwijl een molecule een typische afmeting van
                \[
                1\cdot10^{-9}\,\mathrm m
                \]
                heeft.
                
                Hoeveel keer groter is de typische afmeting van de cel?
                
                \[
                \answer{10000}
                \]
                
                \begin{oplossing}
                \[
                \dfrac{10^{-5}}{10^{-9}}
                =10^{-5-(-9)}
                =10^4
                =10\,000.
                \]
                
                De cel is op deze schaal dus ongeveer $10\,000$ keer groter.
                \end{oplossing}
                \end{question}
                
                
                \begin{question}
                Hoeveel ordes van grootte verschillen de typische afmetingen van de cel
                en de molecule uit de vorige vraag?
                
                Antwoord: $\answer{4}$ ordes van grootte.
                
                \begin{oplossing}
                De exponenten zijn $-5$ en $-9$.
                
                Hun verschil is
                \[
                -5-(-9)=4.
                \]
                
                De afmetingen verschillen dus vier ordes van grootte.
                \end{oplossing}
                \end{question}
                
                
                \begin{question}
                Een leerling zegt:
                
                ``Een grootheid die $3$ ordes van grootte groter is, is $3$ keer zo groot.''
                
                Is deze uitspraak correct? Leg uit.
                
                \begin{oplossing}
                Nee.
                
                Eén orde van grootte komt overeen met een factor $10$.
                Drie ordes van grootte komen dus overeen met
                \[
                10^3=1000.
                \]
                
                Een grootheid die drie ordes van grootte groter is, is dus ongeveer
                $1000$ keer zo groot, niet $3$ keer zo groot.
                \end{oplossing}
                \end{question}
                
                
                \begin{question}
                Leg uit waarom wetenschappelijke notatie bijzonder geschikt is om
                zeer grote en zeer kleine wetenschappelijke grootheden te vergelijken.
                
                \begin{oplossing}
                In wetenschappelijke notatie
                \[
                a\cdot10^n
                \]
                maakt de exponent $n$ onmiddellijk zichtbaar op welke schaal een
                grootheid ligt.
                
                Daardoor kun je snel ordegroottes vergelijken en zie je bijvoorbeeld
                onmiddellijk dat $10^{-9}$ veel kleiner is dan $10^{-5}$.
                
                Ook producten, quotiënten en verhoudingen zijn eenvoudiger te berekenen
                omdat je rechtstreeks met de machten van $10$ kunt rekenen.
                \end{oplossing}
                \end{question}
                
                \end{multicols}
                
                \end{oefening}

    \begin{oefening}[Twee keer twintig procent]{\oefU\ESS}
    \\
    
    Een toestel kost €500. De prijs stijgt eerst met $20\%$ en daalt daarna met $20\%$.
    
    \begin{multicols}{2}
    
    \begin{question}
    Bereken de prijs na de stijging van $20\%$.
    
    €$\answer{600}$
    
    \begin{oplossing}
    Een stijging met $20\%$ komt overeen met groeifactor
    \[
    1+\dfrac{20}{100}=1{,}20.
    \]
    
    Dus
    \[
    500\cdot1{,}20
    =600.
    \]
    \end{oplossing}
    \end{question}
    
    
    \begin{question}
    Bereken de uiteindelijke prijs na de daaropvolgende daling van $20\%$.
    
    €$\answer{480}$
    
    \begin{oplossing}
    Een daling met $20\%$ komt overeen met groeifactor
    \[
    1-\dfrac{20}{100}=0{,}80.
    \]
    
    Dus
    \[
    600\cdot0{,}80
    =480.
    \]
    \end{oplossing}
    \end{question}
    
    
    \begin{question}
    Met welke totale factor is de oorspronkelijke prijs vermenigvuldigd?
    
    $\answer{0.96}$
    
    \begin{oplossing}
    De twee opeenvolgende groeifactoren zijn
    \[
    1{,}20
    \qquad\text{en}\qquad
    0{,}80.
    \]
    
    De totale factor is
    \[
    1{,}20\cdot0{,}80
    =0{,}96.
    \]
    \end{oplossing}
    \end{question}
    
    
    \begin{question}
    Wat is de totale procentuele verandering ten opzichte van de oorspronkelijke prijs?
    
    $\answer{-4}\%$
    
    \begin{oplossing}
    De totale groeifactor is
    \[
    0{,}96.
    \]
    
    Daarom is de procentuele verandering
    \[
    (0{,}96-1)\cdot100\%
    =-4\%.
    \]
    
    De prijs is uiteindelijk dus $4\%$ lager.
    \end{oplossing}
    \end{question}
    
    
    \begin{question}
    Leg uit waarom de prijs niet opnieuw €500 bedraagt.
    
    \begin{oplossing}
    De stijging en de daling worden niet op hetzelfde bedrag berekend.
    
    De stijging bedraagt
    \[
    20\%\text{ van }500=100,
    \]
    terwijl de daling bedraagt
    \[
    20\%\text{ van }600=120.
    \]
    
    Dezelfde procentuele stijging en daling heffen elkaar dus niet op.
    \end{oplossing}
    \end{question}
    
    
    \begin{question}
    Welke procentuele daling na de stijging is nodig om exact bij €500 uit te komen?
    
    $\answer{16.6666667}\%$
    
    \begin{oplossing}
    Na de stijging kost het toestel €600.
    
    Om terug te keren naar €500 moet de prijs met €100 dalen. Ten opzichte van €600 is dat
    \[
    \dfrac{100}{600}\cdot100\%
    =16\tfrac23\%.
    \]
    
    Er is dus een daling van
    \[
    16\tfrac23\%
    \]
    nodig.
    \end{oplossing}
    \end{question}
    
    
    \begin{question}
    Een andere prijs stijgt eerst met $10\%$ en daarna nog eens met $10\%$.
    
    Met welke totale factor wordt de prijs vermenigvuldigd?
    
    $\answer{1.21}$
    
    \begin{oplossing}
    Elke stijging met $10\%$ heeft groeifactor $1{,}10$.
    
    Dus
    \[
    1{,}10\cdot1{,}10
    =1{,}21.
    \]
    
    De totale groeifactor is $1{,}21$.
    \end{oplossing}
    \end{question}
    
    
    \begin{question}
    Met hoeveel procent is de prijs uit de vorige vraag in totaal gestegen?
    
    $\answer{21}\%$
    
    \begin{oplossing}
    De totale groeifactor is
    \[
    1{,}21.
    \]
    
    Daarom is de totale procentuele stijging
    \[
    (1{,}21-1)\cdot100\%
    =21\%.
    \]
    \end{oplossing}
    \end{question}
    
    
    \begin{question}
    Een hoeveelheid daalt eerst met $10\%$ en stijgt daarna met $10\%$.
    
    Is de eindwaarde groter dan, kleiner dan of gelijk aan de beginwaarde? Bereken de totale procentuele verandering.
    
    \begin{oplossing}
    De totale groeifactor is
    \[
    0{,}90\cdot1{,}10
    =0{,}99.
    \]
    
    Dus
    \[
    (0{,}99-1)\cdot100\%
    =-1\%.
    \]
    
    De eindwaarde is dus $1\%$ kleiner dan de beginwaarde.
    \end{oplossing}
    \end{question}
    
    
    \begin{question}
    Een leerling beweert:
    
    ``Een stijging met $p\%$ gevolgd door een daling met $p\%$ brengt je altijd terug naar de beginwaarde.''
    
    Is deze uitspraak correct? Verklaar algemeen.
    
    \begin{oplossing}
    Nee.
    
    Een stijging met $p\%$ heeft groeifactor
    \[
    1+\dfrac{p}{100},
    \]
    en een daling met $p\%$ heeft groeifactor
    \[
    1-\dfrac{p}{100}.
    \]
    
    De totale factor is
    \[
    \left(1+\dfrac{p}{100}\right)
    \left(1-\dfrac{p}{100}\right).
    \]
    
    Met het verschil van twee kwadraten:
    \[
    1-\left(\dfrac{p}{100}\right)^2.
    \]
    
    Voor $p\neq0$ is deze factor kleiner dan $1$.
    
    De eindwaarde is dus kleiner dan de beginwaarde.
    \end{oplossing}
    \end{question}
    
    \end{multicols}
    
    \end{oefening}

\begin{oplossing}

\]
\[

\]

Kwadrateren geeft
\[
\frac{T^2}{4\pi^2}
=\frac{\ell}{g}.
\]

Dus
\[
\ell=\frac{gT^2}{4\pi^2}.
\]
\end{oplossing}
\end{question}


\begin{question}
Bereken $\ell$ als $T=2{,}0\,\mathrm s$ en
$g=9{,}81\,\mathrm{m/s^2}$.

\[
\ell\approx\answer{0.994}\,\mathrm m
\]

\begin{oplossing}
\[
\ell
=\frac{9{,}81\cdot(2{,}0)^2}{4\pi^2}
\]
\[
=\frac{39{,}24}{4\pi^2}
\approx0{,}994\,\mathrm m.
\]
\end{oplossing}
\end{question}


\begin{question}
Controleer de eenheid van het resultaat.

\begin{oplossing}
Uit
\[
\ell=\frac{gT^2}{4\pi^2}
\]
volgt voor de eenheden:
\[
\frac{\mathrm m}{\mathrm s^2}
\cdot\mathrm s^2
=\mathrm m.
\]

Dat is correct voor een lengte.
\end{oplossing}
\end{question}


\begin{question}
Wat gebeurt er met de periode $T$ als de lengte $\ell$ vier keer zo groot wordt?

\begin{oplossing}
Uit
\[
T=2\pi\sqrt{\frac{\ell}{g}}
\]
volgt dat $T$ evenredig is met $\sqrt{\ell}$.

Als
\[
\ell\to4\ell,
\]
dan
\[
T\to\sqrt4\,T=2T.
\]

De periode verdubbelt.
\end{oplossing}
\end{question}


\begin{question}

\begin{oplossing}

\[
T\propto\frac1{\sqrt g},
\]
geldt bij
\[
g\to4g:
\]
\[
T\to\frac{T}{\sqrt4}
=\frac T2.
\]

De periode halveert.
\end{oplossing}
\end{question}


\begin{question}
Een slinger heeft lengte $1{,}00\,\mathrm m$.

\]

\begin{oplossing}
\[
T
=2\pi\sqrt{\frac{1{,}00}{9{,}81}}

\]
\end{oplossing}
\end{question}

    Omdat $P(3,10)$ op de grafiek ligt:
    \[
    10=3a+4
    \]
    \[
    3a=6
    \]
    dus
    \[
    a=2.
    \]
    \end{oplossing}
    \end{question}
    
    
    \begin{question}
    Bepaal het snijpunt met de $x$-as.
    
    \begin{oplossing}
    Met $a=2$ is
    \[
    f_a(x)=2x+4.
    \]
    
    Voor het snijpunt met de $x$-as geldt
    \[
    2x+4=0.
    \]
    
    Dus
    \[
    x=-2.
    \]
    
    Het snijpunt is
    \[
    \boxed{(-2,0)}.
    \]
    \end{oplossing}
    \end{question}
    
    
    \begin{question}
    Voor welke $x$ geldt $f_a(x)>0$?
    
    \begin{oplossing}
    \[
    2x+4>0
    \]
    \[
    2x>-4
    \]
    dus
    \[
    x>-2.
    \]
    
    Daarom
    \[
    x\in]-2,+\infty[.
    \]
    \end{oplossing}
    \end{question}
    
    
    \begin{question}

    \begin{oplossing}

    \[
    x=-2.

    Rechts van dit snijpunt ligt de grafiek boven de $x$-as. Daar geldt dus
    \[
    f_a(x)>0.

    \end{oplossing}

    f_a(x)=3x-1?

    x=5.

    g(x)=bx-2
    \]

    b=\answer{2}

    \[
    g(x)=bx-2

    dezelfde rechte is als $f_a$? Motiveer.

    Nee.

    \[
    a>0
    \quad\Rightarrow\quad

    a<0
    \quad\Rightarrow\quad
    f_a \text{ is dalend},

    \[
    a=0
    \quad\Rightarrow\quad
    f_a \text{ is constant}.
    \]
    \end{oplossing}
    \end{question}

    \begin{oefening}[Ontwerpen met voorwaarden]{\oefG}
    \\
    
    Een rechthoekig zonnepaneel heeft oppervlakte $1{,}80\,\mathrm{m^2}$.
    
    De lengte is $0{,}60\,\mathrm m$ groter dan de breedte.
    
    \begin{center}
    \begin{tikzpicture}[scale=0.9]
    \draw[very thick] (0,0) rectangle (4.2,2.5);
    
    \node[below] at (2.1,0) {$x+0{,}60$};
    \node[left] at (0,1.25) {$x$};
    
    \draw[dashed] (0,0)--(4.2,2.5);
    \node[above left] at (2.2,1.35) {$d$};
    \end{tikzpicture}
    \end{center}
    
    \begin{multicols}{2}
    
    \begin{question}
    Stel een vergelijking op voor de breedte.
    
    \begin{oplossing}
    Noem de breedte
    \[
    x\,\mathrm m.
    \]
    
    Dan is de lengte
    \[
    x+0{,}60.
    \]
    
    Uit de oppervlaktevoorwaarde volgt
    \[
    x(x+0{,}60)=1{,}80.
    \]
    \end{oplossing}
    \end{question}
    
    
    \begin{question}
    Bepaal de afmetingen.
    
    \begin{oplossing}
    \[
    x^2+0{,}60x-1{,}80=0.
    \]
    
    Met de nulpuntenformule:
    \[
    x=
    \frac{-0{,}60\pm\sqrt{0{,}36+7{,}20}}{2}
    \]
    \[
    =
    \frac{-0{,}60\pm\sqrt{7{,}56}}{2}.
    \]
    
    Alleen de positieve oplossing is fysisch zinvol:
    \[
    x\approx1{,}0748\,\mathrm m.
    \]
    
    De lengte is
    \[
    x+0{,}60
    \approx1{,}6748\,\mathrm m.
    \]
    \end{oplossing}
    \end{question}
    
    
    \begin{question}
    Bereken de diagonaal.
    
    \begin{oplossing}
    \[
    d
    =
    \sqrt{x^2+(x+0{,}60)^2}.
    \]
    
    Invullen geeft
    \[
    d
    \approx
    \sqrt{1{,}0748^2+1{,}6748^2}
    \]
    \[
    \approx1{,}990\,\mathrm m.
    \]
    \end{oplossing}
    \end{question}
    
    
    \begin{question}
    Een transportkist laat maximaal een diagonaal van
    $1{,}75\,\mathrm m$ toe.
    
    Past het paneel? Motiveer.
    
    \begin{oplossing}
    Nee.
    
    De diagonaal is ongeveer
    \[
    1{,}99\,\mathrm m,
    \]
    en dus groter dan
    \[
    1{,}75\,\mathrm m.
    \]
    \end{oplossing}
    \end{question}
    
    
    \begin{question}
    Hoeveel procent groter is de diagonaal dan de maximale toegelaten diagonaal?
    
    \begin{oplossing}
    Het verschil is ongeveer
    \[
    1{,}990-1{,}75
    =0{,}240\,\mathrm m.
    \]
    
    Relatief ten opzichte van de maximale toegelaten diagonaal:
    \[
    \frac{0{,}240}{1{,}75}\cdot100\%
    \approx13{,}7\%.
    \]
    
    De diagonaal is dus ongeveer $13{,}7\%$ te groot.
    \end{oplossing}
    \end{question}
    
    
    \begin{question}
    De fabrikant kan de lengte en breedte aanpassen, maar de oppervlakte moet
    $1{,}80\,\mathrm{m^2}$ blijven.
    
    Welke vorm geeft bij vaste oppervlakte de kleinste diagonaal?
    
    \begin{oplossing}
    Noem de zijden $x$ en $y$.
    
    Omdat
    \[
    xy=1{,}80,
    \]
    is
    \[
    y=\frac{1{,}80}{x}.
    \]
    
    Voor de diagonaal:
    \[
    d^2=x^2+y^2
    =x^2+\frac{1{,}80^2}{x^2}.
    \]
    
    Bij vaste oppervlakte is deze som minimaal wanneer
    \[
    x=y.
    \]
    
    De rechthoek moet dus een vierkant zijn.
    \end{oplossing}
    \end{question}
    
    
    \begin{question}
    Bereken de zijde en de minimale diagonaal van dat vierkante paneel.
    
    \begin{oplossing}
    Voor een vierkant met oppervlakte $1{,}80\,\mathrm{m^2}$:
    \[
    x^2=1{,}80.
    \]
    
    Dus
    \[
    x=\sqrt{1{,}80}
    \approx1{,}342\,\mathrm m.
    \]
    
    De diagonaal is
    \[
    d=x\sqrt2
    \]
    dus
    \[
    d=\sqrt{1{,}80}\sqrt2
    =\sqrt{3{,}60}
    \approx1{,}897\,\mathrm m.
    \]
    
    Ook dit is nog groter dan $1{,}75\,\mathrm m$.
    \end{oplossing}
    \end{question}
    
    
    \begin{question}
    Kan er überhaupt een rechthoekig paneel met oppervlakte
    $1{,}80\,\mathrm{m^2}$ in een kist met maximale diagonaal
    $1{,}75\,\mathrm m$ passen?
    
    \begin{oplossing}
    Nee.
    
    De kleinste mogelijke diagonaal bij vaste oppervlakte ontstaat bij een vierkant.
    
    Die minimale diagonaal is
    \[
    d_{\min}\approx1{,}897\,\mathrm m.
    \]
    
    Omdat
    \[
    1{,}897>1{,}75,
    \]
    kan geen enkele rechthoek met oppervlakte $1{,}80\,\mathrm{m^2}$ voldoen.
    \end{oplossing}
    \end{question}
    
    \end{multicols}
    
    \end{oefening}

    \begin{oefening}[Model kritisch beoordelen]{\oefG}
    \\
    
    Een sensor wordt tussen twee kalibratiepunten lineair gemodelleerd:
    \[
    (0,0{,}20)
    \qquad\text{en}\qquad
    (100,4{,}80).
    \]
    
    \begin{multicols}{2}
    
    \begin{question}
    Stel het lineaire model op.
    
    \begin{oplossing}
    De rico is
    \[
    m=
    \frac{4{,}80-0{,}20}{100-0}
    =\frac{4{,}60}{100}
    =0{,}046.
    \]
    
    Omdat de sensorwaarde bij invoer $0$ gelijk is aan $0{,}20$:
    \[
    \boxed{y=0{,}046x+0{,}20}.
    \]
    \end{oplossing}
    \end{question}
    
    
    \begin{question}
    Bereken de voorspelde sensorwaarde bij invoer $60$.
    
    \[
    \answer{2.96}
    \]
    
    \begin{oplossing}
    \[
    y(60)
    =0{,}046\cdot60+0{,}20
    =2{,}96.
    \]
    \end{oplossing}
    \end{question}
    
    
    \begin{question}
    Bereken welke invoer hoort bij sensorwaarde $3{,}00$.
    
    \[
    x\approx\answer{60.87}
    \]
    
    \begin{oplossing}
    \[
    3{,}00
    =0{,}046x+0{,}20
    \]
    \[
    2{,}80
    =0{,}046x
    \]
    dus
    \[
    x\approx60{,}87.
    \]
    \end{oplossing}
    \end{question}
    
    
    \begin{question}
    Interpreteer de richtingscoëfficiënt $0{,}046$ in de context.
    
    \begin{oplossing}
    Voor elke toename van de invoer met $1$ eenheid voorspelt het lineaire model dat de sensorwaarde gemiddeld met
    \[
    0{,}046
    \]
    toeneemt.
    \end{oplossing}
    \end{question}
    
    
    \begin{question}
    Interpreteer het getal $0{,}20$.
    
    \begin{oplossing}
    Volgens het model is
    \[
    y(0)=0{,}20.
    \]
    
    De sensor geeft dus bij nul invoer nog een uitgangswaarde van $0{,}20$.
    
    Dat kan bijvoorbeeld overeenkomen met een offset in de sensor.
    \end{oplossing}
    \end{question}
    
    
    \begin{question}
    Het model voorspelt ook een waarde bij invoer $1000$.
    
    Bereken die waarde.
    
    \begin{oplossing}
    \[
    y(1000)
    =0{,}046\cdot1000+0{,}20
    =46{,}20.
    \]
    \end{oplossing}
    \end{question}
    
    
    \begin{question}
    Leg uit waarom de vorige waarde niet automatisch een betrouwbare fysische voorspelling is.
    
    \begin{oplossing}
    De kalibratie gebeurde slechts tussen invoer $0$ en $100$.
    
    Bij invoer $1000$ extrapoleren we zeer ver buiten het gekalibreerde gebied.
    
    Een echte sensor kan daar bijvoorbeeld:
    \begin{itemize}
    \item verzadigen;
    \item niet-lineair worden;
    \item beschadigd raken;
    \item buiten zijn meetbereik vallen.
    \end{itemize}
    
    De waarde $46{,}20$ is dus wel een correcte uitkomst van het lineaire model, maar niet noodzakelijk een betrouwbare voorspelling van de echte sensor.
    \end{oplossing}
    \end{question}
    
    
    \begin{question}
    Bij invoer $80$ wordt experimenteel een sensorwaarde $3{,}70$ gemeten.
    
    Bereken de voorspelling van het model en de absolute afwijking.
    
    \begin{oplossing}
    Het model voorspelt
    \[
    y(80)
    =0{,}046\cdot80+0{,}20
    =3{,}88.
    \]
    
    De absolute afwijking is
    \[
    |3{,}70-3{,}88|
    =0{,}18.
    \]
    \end{oplossing}
    \end{question}
    
    
    \begin{question}
    Bereken de relatieve afwijking van de gemeten waarde ten opzichte van de modelvoorspelling.
    
    \begin{oplossing}
    De absolute afwijking is
    \[
    0{,}18.
    \]
    
    Relatief ten opzichte van de voorspelling:
    \[
    \frac{0{,}18}{3{,}88}\cdot100\%
    \approx4{,}64\%.
    \]
    
    De meting wijkt dus ongeveer $4{,}6\%$ af van de modelvoorspelling.
    \end{oplossing}
    \end{question}
    
    
    \begin{question}
    Een tweede kalibratiemeting bij invoer $50$ geeft $2{,}40$.
    
    Onderzoek of dit punt exact op het lineaire model ligt.
    
    \begin{oplossing}
    Volgens het model:
    \[
    y(50)
    =0{,}046\cdot50+0{,}20
    =2{,}50.
    \]
    
    De gemeten waarde is
    \[
    2{,}40.
    \]
    
    Het punt ligt dus niet exact op het model.
    
    Het verschil bedraagt
    \[
    0{,}10.
    \]
    \end{oplossing}
    \end{question}
    
    
    \begin{question}
    Wat zou je met meerdere kalibratiepunten doen in plaats van zomaar een rechte door twee punten te trekken?
    
    \begin{oplossing}
    Met meerdere meetpunten is het meestal beter om een regressiemodel te bepalen.
    
    Bij een ongeveer lineair verband kan men een lineaire regressie gebruiken waarbij de rechte zo goed mogelijk bij alle meetpunten aansluit.
    
    Daarbij kan men ook de residuen onderzoeken om na te gaan of een lineair model werkelijk geschikt is.
    \end{oplossing}
    \end{question}
    
    \end{multicols}

    \begin{oefening}[Twee modellen vergelijken]{\oefG}

    Twee lineaire modellen beschrijven de temperatuur $T$ van twee processen als functie van de tijd $t$:
    
    \[
    T_1(t)=18+2{,}4t
    \]
    en
    \[
    T_2(t)=30+1{,}2t.

    \begin{oplossing}
    Bij $t=0$:
    \[
    T_1(0)=18
    \]
    en
    \[
    T_2(0)=30.
    \]
    
    Proces $1$ start dus bij $18^\circ\mathrm C$ en proces $2$ bij $30^\circ\mathrm C$.
    \end{oplossing}
    \end{question}
    
    
    \begin{question}
    Interpreteer de richtingscoëfficiënten.
    
    \begin{oplossing}
    Voor proces $1$ stijgt de temperatuur met
    \[
    2{,}4^\circ\mathrm C
    \]
    per minuut.
    
    Voor proces $2$ stijgt de temperatuur met
    \[
    1{,}2^\circ\mathrm C
    \]
    per minuut.
    \end{oplossing}
    \end{question}

    t=\answer{10}

    =

    en
    \[
    t=10.

    T_1(10)

    =42.

    \[

    \]

    >

    \]
    op.
    
    Dus
    \[
    1{,}2t>12
    \]
    en
    \[
    t>10.

    \[
    t>10
    \]
    ligt de grafiek van $T_1$ boven de grafiek van $T_2$.
    \end{oplossing}
    \end{question}

    \begin{oefening}[Van gegevens naar model]{\oefG}
    \\
    
    Tijdens een experiment wordt de afstand $s$ van een bewegend voorwerp gemeten.
    
    \[
    \begin{array}{c|cccc}
    t\;(\mathrm s) & 1 & 3 & 5 & 7\\
    \hline
    s\;(\mathrm m) & 4{,}1 & 8{,}2 & 12{,}0 & 16{,}3
    \end{array}
    \]
    
    \begin{multicols}{2}
    
    \begin{question}
    Onderzoek of de gegevens exact lineair zijn.
    
    \begin{oplossing}
    Bij gelijke tijdsstappen van $2\,\mathrm s$ zijn de verschillen in afstand:
    \[
    8{,}2-4{,}1=4{,}1,
    \]
    \[
    12{,}0-8{,}2=3{,}8,
    \]
    \[
    16{,}3-12{,}0=4{,}3.
    \]
    
    De verschillen zijn niet exact gelijk.
    
    De gegevens zijn dus niet exact lineair.
    \end{oplossing}
    \end{question}
    
    
    \begin{question}
    Leg uit waarom een lineair model toch redelijk kan zijn.
    
    \begin{oplossing}
    De verschillen liggen allemaal dicht bij $4\,\mathrm m$ per $2\,\mathrm s$.
    
    De afwijkingen kunnen veroorzaakt worden door meetonzekerheid.
    
    Een lineair model kan daarom een redelijke benadering zijn.
    \end{oplossing}
    \end{question}
    
    
    \begin{question}
    Schat de gemiddelde snelheid uit de eerste en laatste meting.
    
    \begin{oplossing}
    \[
    v_{\mathrm{gem}}
    =
    \frac{16{,}3-4{,}1}{7-1}
    \]
    \[
    =
    \frac{12{,}2}{6}
    \approx2{,}03\,\mathrm{m/s}.
    \]
    \end{oplossing}
    \end{question}
    
    
    \begin{question}
    Stel met de eerste en laatste meting een lineair model
    \[
    s(t)=vt+s_0
    \]
    op.
    
    \begin{oplossing}
    De rico is
    \[
    v
    =
    \frac{16{,}3-4{,}1}{7-1}
    \approx2{,}033.
    \]
    
    Met $(1,4{,}1)$:
    \[
    4{,}1
    =
    2{,}033\cdot1+s_0.
    \]
    
    Dus
    \[
    s_0\approx2{,}067.
    \]
    
    Een mogelijk model is
    \[
    \boxed{s(t)\approx2{,}033t+2{,}067}.
    \]
    \end{oplossing}
    \end{question}
    
    
    \begin{question}
    Bereken met dit model de voorspelde afstand bij $t=5\,\mathrm s$ en vergelijk met de meting.
    
    \begin{oplossing}
    Volgens het model:
    \[
    s(5)
    \approx2{,}033\cdot5+2{,}067
    \]
    \[
    \approx12{,}23\,\mathrm m.
    \]
    
    De gemeten waarde is
    \[
    12{,}0\,\mathrm m.
    \]
    
    Het verschil is dus ongeveer
    \[
    0{,}23\,\mathrm m.
    \]
    \end{oplossing}
    \end{question}
    
    
    \begin{question}
    Waarom is een model dat exact door alle meetpunten gaat niet noodzakelijk beter?
    
    \begin{oplossing}
    Meetgegevens bevatten meestal meetonzekerheid en toevallige afwijkingen.
    
    Een model dat exact door elk meetpunt gaat, kan daardoor vooral de meetruis beschrijven in plaats van het onderliggende fysische verband.
    
    Een eenvoudiger model dat de algemene trend goed weergeeft, kan wetenschappelijk betekenisvoller zijn.
    \end{oplossing}
    \end{question}
    
    \end{multicols}
    
    \end{oefening}
