\begin{oefening}[Teken en sterkte]{\oefB \ESS}
Geef voor elke correlatiecoëfficiënt de richting en een globale
beschrijving van de sterkte van het lineaire verband.

\[
\begin{array}{c|ccccc}
& A&B&C&D&E\\ \hline
r&0{,}94&-0{,}88&0{,}12&-0{,}41&1
\end{array}
\]

\begin{oplossing}
\begin{itemize}
  \item $A$: sterk positief lineair;
  \item $B$: sterk negatief lineair;
  \item $C$: zwak positief lineair;
  \item $D$: eerder zwak negatief lineair;
  \item $E$: perfect positief lineair.
\end{itemize}
De woorden ``sterk'' en ``zwak'' zijn globale beschrijvingen; de
context en de puntenwolk blijven belangrijk.
\end{oplossing}
\end{oefening}


\begin{oefening}[Welke is sterker?]{\oefB \ESS}
Vergelijk telkens de sterkte van de lineaire samenhang.

\begin{question}
Welke is sterker?

\[
r_1=-0{,}91,
\qquad
r_2=0{,}84
\]

\begin{multipleChoice}
  \choice[correct]{$r_1$}
  \choice{$r_2$}
  \choice{even sterk}
\end{multipleChoice}
\end{question}

\begin{question}
Waarom?

\begin{oplossing}
Omdat $|-0{,}91|=0{,}91$ groter is dan $|0{,}84|=0{,}84$.
Het minteken geeft alleen de richting aan.
\end{oplossing}
\end{question}

\begin{question}
Welke is sterker?

\[
r_3=-0{,}72,
\qquad
r_4=0{,}72
\]

\begin{multipleChoice}
  \choice{$r_3$}
  \choice{$r_4$}
  \choice[correct]{even sterk}
\end{multipleChoice}
\end{question}
\end{oefening}


\begin{oefening}[Mogelijk of onmogelijk?]{\oefB \ESS}
Duid aan welke waarden een correlatiecoëfficiënt kunnen zijn.

\[
-1{,}4,\quad -1,\quad -0{,}37,\quad 0,\quad 0{,}83,\quad 1,\quad 1{,}02
\]

\begin{oplossing}
Mogelijk zijn

\[
-1,\quad -0{,}37,\quad 0,\quad 0{,}83,\quad 1.
\]

Een correlatiecoëfficiënt ligt steeds tussen $-1$ en $1$.
\end{oplossing}
\end{oefening}


\begin{oefening}[Wat betekent nul?]{\oefB \ESS}
Een dataset heeft

\[
r=0{,}02.
\]

Een leerling besluit:

\begin{quote}
Er bestaat geen verband tussen de twee grootheden.
\end{quote}

Beoordeel deze uitspraak.

\begin{oplossing}
De uitspraak is te sterk. $r=0{,}02$ wijst op vrijwel geen
\emph{lineaire} samenhang. Er kan nog steeds een duidelijk niet-lineair
verband bestaan. Daarom moet ook de puntenwolk bekeken worden.
\end{oplossing}
\end{oefening}





\begin{oefening}[Wiskunde en fysica]{\oefU \ESS}

Gebruik

\csv
{https://raw.githubusercontent.com/Ingmar-1986/wiskunde_cursus_2/main/modules/Data/Statistiek/wiskfys.csv}
{wiskfys.csv}

Open de gegevens met geschikte ICT. Maak een spreidingsdiagram en
bepaal de correlatiecoëfficiënt.

\begin{question}
Beschrijf eerst de puntenwolk zonder naar $r$ te kijken.

\begin{oplossing}
De puntenwolk vertoont een positieve, ongeveer lineaire samenhang met
duidelijke spreiding.
\end{oplossing}
\end{question}

\begin{question}
ICT geeft ongeveer
\[
r=0{,}743.
\]
Leg uit hoe teken en absolute waarde van $r$ overeenkomen met de grafiek.

\begin{oplossing}
$r$ is positief, wat overeenkomt met de stijgende tendens.
Omdat $|r|$ duidelijk groter is dan nul maar niet vlak bij $1$ ligt,
is er een behoorlijk sterke maar niet perfecte lineaire samenhang.
Dat past bij de zichtbare spreiding rond een stijgende band.
\end{oplossing}
\end{question}

\begin{question}
Een leerling besluit:
\begin{quote}
``Goede punten voor wiskunde veroorzaken goede punten voor fysica.''
\end{quote}
Waarom volgt dit niet uit $r$?

\begin{oplossing}
$r$ beschrijft lineaire samenhang, geen causaliteit. Andere factoren
kunnen beide resultaten beïnvloeden.
\end{oplossing}
\end{question}

\end{oefening}


\begin{oefening}[Geschiedenis en aardrijkskunde]{\oefU \ESS}

Gebruik

\csv
{https://raw.githubusercontent.com/Ingmar-1986/wiskunde_cursus_2/main/modules/Data/Statistiek/examen.csv}
{examen.csv}

Maak een spreidingsdiagram en bepaal $r$.

\begin{question}
ICT geeft ongeveer
\[
r=-0{,}107.
\]
Wat betekent deze waarde?

\begin{oplossing}
Er is in deze dataset weinig lineaire samenhang tussen de twee
examenscores. Het licht negatieve teken verandert daar weinig aan.
\end{oplossing}
\end{question}

\begin{question}
Waarom moet je de puntenwolk toch bekijken en niet alleen $r$ rapporteren?

\begin{oplossing}
$r$ vat slechts lineaire samenhang samen. De grafiek kan uitschieters,
clusters of een niet-lineaire structuur tonen die in één getal
verdwijnt.
\end{oplossing}
\end{question}

\end{oefening}


\begin{oefening}[Druk en volume: sterk maar niet perfect lineair]{\oefU \ESS}

Gebruik opnieuw

\csv
{https://raw.githubusercontent.com/Ingmar-1986/wiskunde_cursus_2/main/modules/Data/Statistiek/druk.csv}
{druk.csv}

\begin{question}
Bepaal met ICT $r$.

\begin{oplossing}
\[
r\approx -0{,}971.
\]
\end{oplossing}
\end{question}

\begin{question}
Een leerling zegt:
\begin{quote}
``Omdat $|r|$ bijna $1$ is, is een rechte zeker het beste model.''
\end{quote}
Beoordeel deze uitspraak aan de hand van de puntenwolk.

\begin{oplossing}
De uitspraak is te sterk. $r$ is inderdaad zeer groot in absolute
waarde, maar de puntenwolk vertoont een duidelijke kromming. Een hoge
lineaire correlatie garandeert niet dat een rechte inhoudelijk het
beste model is.
\end{oplossing}
\end{question}

\end{oefening}


\begin{oefening}[Vaders en zonen]{\oefU}

Gebruik

\csv
{https://raw.githubusercontent.com/Ingmar-1986/wiskunde_cursus_2/main/modules/Data/Statistiek/vaderzoon.csv}
{vaderzoon.csv}

Maak een spreidingsdiagram en bepaal $r$.

\begin{question}
Beschrijf de lineaire samenhang.

\begin{oplossing}
De correlatie is positief en matig tot behoorlijk sterk:
\[
r\approx0{,}704.
\]
Langere vaders hebben in deze dataset doorgaans langere zonen, maar
de samenhang is verre van perfect.
\end{oplossing}
\end{question}

\begin{question}
Waarom mag je uit $r$ geen exacte voorspelling voor één individuele
zoon afleiden?

\begin{oplossing}
De punten vertonen duidelijke spreiding. Correlatie beschrijft een
globale samenhang in de groep en bepaalt geen individuele uitkomst.
\end{oplossing}
\end{question}

\end{oefening}




\begin{oefening}[Correlaties vergelijken zonder de grafiek te vergeten]{\oefU \ESS}

Vergelijk de datasets

\begin{itemize}
  \item \texttt{examen.csv},
  \item \texttt{wiskfys.csv},
  \item \texttt{schoenmaat.csv},
  \item \texttt{druk.csv}.
\end{itemize}

Gebruik ICT om voor elke dataset een spreidingsdiagram en $r$ te verkrijgen.

\begin{question}
Rangschik de datasets volgens $|r|$.

\begin{oplossing}
Met de huidige datasets geldt ongeveer:
\[
|r_{\text{examen}}|\approx0{,}107,\qquad
|r_{\text{wiskfys}}|\approx0{,}743,
\]
\[
|r_{\text{schoenmaat}}|\approx0{,}895,\qquad
|r_{\text{druk}}|\approx0{,}971.
\]
\end{oplossing}
\end{question}

\begin{question}
Welke dataset toont het duidelijkst waarom de grootste $|r|$ niet
automatisch betekent dat een lineair model het meest geschikt is?

\begin{oplossing}
\texttt{druk.csv}. De absolute correlatie is zeer groot, maar de
puntenwolk vertoont een systematisch krom patroon.
\end{oplossing}
\end{question}

\begin{question}
Formuleer één algemene regel voor het interpreteren van $r$.

\begin{oplossing}
Interpreteer $r$ nooit los van de puntenwolk en de context. $r$ meet
alleen de richting en sterkte van \emph{lineaire} samenhang.
\end{oplossing}
\end{question}

\end{oefening}



\begin{oefening}[Kies het betere argument]{\oefG}

Twee leerlingen vergelijken twee datasets.

Leerling A zegt:
\begin{quote}
``Dataset 1 is beter voor regressie, want $|r|$ is groter.''
\end{quote}

Leerling B zegt:
\begin{quote}
``Ik wil eerst beide puntenwolken zien en daarna $r$ interpreteren.''
\end{quote}

Schrijf een beargumenteerde reactie. Gebruik minstens één concrete
dataset uit dit hoofdstuk als tegenvoorbeeld tegen een te mechanische
interpretatie van $r$.

\begin{oplossing}
Leerling B redeneert statistisch sterker. Een grote $|r|$ is relevante
informatie, maar toont geen kromming, uitschieters of clusters.
\texttt{druk.csv} is een duidelijk tegenvoorbeeld: $|r|$ ligt dicht
bij $1$, terwijl de puntenwolk een krom patroon vertoont.
\end{oplossing}

\end{oefening}
