}{}


% ============================================================
% HANDMATIGE HOOFDSTUKGEGEVENS
% ============================================================
%
% Alleen nodig wanneer een hoofdstuk afzonderlijk wordt
% gecompileerd en je toch een thema- en hoofdstuknummer
% wilt tonen.
%
% Gebruik vóór \xmtitle:
%
% \fvdchapterdata
%     {1}
%     {Getallen en bewerkingen}
%     {2}
%
% Bij compilatie vanuit index.tex is dit niet nodig.
% ============================================================

\providecommand{\fvdmanualthemanummer}{}
\providecommand{\fvdmanualthematitel}{}
\newif\iffvdmanualchapterdata

\fvdmanualchapterdatafalse

\providecommand{\fvdchapterdata}[3]{%
    \global\fvdmanualchapterdatatrue
    \gdef\fvdmanualthemanummer{#1}%
    \gdef\fvdmanualthematitel{#2}%
    \setcounter{titlenumber}{#3}%
}


% ============================================================
% THEMANUMMER
% ============================================================

\newcommand{\fvdherothemanummer}{%
    \iffvdmanualchapterdata
        \fvdmanualthemanummer
    \else
        \thepart
    \fi
}


% ============================================================
% THEMATITEL
% ============================================================

\newcommand{\fvdherothematitel}{%
    \iffvdmanualchapterdata
        \fvdmanualthematitel
    \else
        \mymoduletitle
    \fi
}


% ============================================================
% HOOFDSTUKNUMMER
% ============================================================

\newcommand{\fvdherohoofdstuknummer}{%
    \arabic{titlenumber}%
}


% ============================================================
% HOOFDSTUKHERO
% ============================================================

\newcommand{\fvdchapterhero}[1]{%

    \thispagestyle{firstpage}%

    \par
    \vspace*{-7mm}

    \noindent
    \begin{tcolorbox}[
        enhanced,
        width=\textwidth,
        boxrule=0pt,
        frame hidden,
        arc=6mm,
        outer arc=6mm,
        left=10mm,
        right=10mm,
        top=7mm,
        bottom=7mm,
        before skip=0pt,
        after skip=8mm,
        interior style={
            left color=fvdHeroNavyLeft,
            right color=fvdHeroNavyRight
        },
        overlay={

            % ------------------------------------------------
            % LICHTE ACHTERGRONDCIRKEL
            % ------------------------------------------------

            \fill[
                fvdHeroCyan,
                opacity=0.035
            ]
            ([xshift=-48mm,yshift=-39mm]frame.north east)
            circle[radius=35mm];


            % ------------------------------------------------
            % SUBTIEL RASTER AAN DE RECHTERKANT
            % ------------------------------------------------

            \begin{scope}[
                opacity=0.055
            ]

                \foreach \x in {0,8,...,72} {
                    \draw[
                        white,
                        line width=0.25pt
                    ]
                    ([xshift=-\x mm]frame.north east)
                    --
                    ([xshift=-\x mm]frame.south east);
                }

                \foreach \y in {0,8,...,64} {
                    \draw[
                        white,
                        line width=0.25pt
                    ]
                    ([yshift=\y mm]frame.south east)
                    --
                    ([xshift=-72mm,yshift=\y mm]frame.south east);
                }

            \end{scope}


            % ------------------------------------------------
            % FORMULE RECHTSBOVEN
            % ------------------------------------------------

            \node[
                anchor=north east,
                text=white,
                opacity=0.11,
                font=\fontsize{10}{12}\selectfont
            ]
            at ([xshift=-12mm,yshift=-7mm]frame.north east)
            {$a^n\cdot a^m=a^{n+m}$};


            % ------------------------------------------------
            % VECTOR RECHTSONDER
            % ------------------------------------------------

            \node[
                anchor=south east,
                text=white,
                opacity=0.09,
                font=\fontsize{10}{12}\selectfont
            ]
            at ([xshift=-16mm,yshift=9mm]frame.south east)
            {$\vec{v}=(x,y,z)$};


            % ------------------------------------------------
            % ASSENSTELSEL MET FUNCTIECURVE
            % ------------------------------------------------

            \begin{scope}[
                shift={
                    ([xshift=-47mm,yshift=-29mm]frame.north east)
                }
            ]

                \draw[
                    white,
                    opacity=0.14,
                    line width=0.35pt
                ]
                (-1.8,0)--(1.8,0);

                \draw[
                    white,
                    opacity=0.14,
                    line width=0.35pt
                ]
                (0,-1.0)--(0,1.1);

                \draw[
                    fvdHeroPink,
                    opacity=0.82,
                    line width=1.1pt,
                    smooth
                ]
                plot coordinates {
                    (-1.65,-0.35)
                    (-1.30,0.35)
                    (-0.90,0.68)
                    (-0.55,0.42)
                    (-0.15,-0.12)
                    (0.25,-0.48)
                    (0.65,-0.35)
                    (1.05,0.22)
                    (1.45,0.72)
                    (1.75,0.98)
                };

            \end{scope}


            % ------------------------------------------------
            % 3D-KUBUS
            % ------------------------------------------------

            \begin{scope}[
                shift={
                    ([xshift=-27mm,yshift=15mm]frame.south east)
                },

            ]

                    fvdHeroCyan,

                    fvdHeroCyan,
                    line width=0.7pt

                    fvdHeroCyan,

                ]
                (1.5,0)--(1.95,0.45);

                \draw[
                    fvdHeroCyan,
                    line width=0.7pt
                ]
                (0,1.5)--(0.45,1.95);

                \draw[
                    fvdHeroCyan,
                    line width=0.7pt
                ]
                (1.5,1.5)--(1.95,1.95);

        }
    ]


    % ========================================================
    % THEMAREGEL
    % ========================================================

    {\sffamily
     \bfseries
     \fontsize{8.5}{10.5}\selectfont
     \color{fvdHeroCyan}

    % ========================================================
    % HOOFDSTUKLABEL
    % ========================================================

    {\sffamily
     \bfseries
     \fontsize{8.5}{10.5}\selectfont

    % ========================================================
    % NUMMER EN TITEL
    % ========================================================

    \noindent
    \begin{minipage}[c]{23mm}

        \begin{tcolorbox}[
            enhanced,
            width=21mm,
            height=21mm,
            valign=center,

            boxrule=0pt,
            frame hidden,
            arc=4.5mm,
            outer arc=4.5mm,
            colback=fvdHeroPink,

             \bfseries
             \fontsize{25}{25}\selectfont

             \fvdherohoofdstuknummer

    \end{minipage}%
    \hspace{3mm}%
    \begin{minipage}[c]{\dimexpr\linewidth-29mm\relax}

        {\raggedright
         \sffamily
         \bfseries
         \fontsize{24}{27}\selectfont
         \color{white}

    % ========================================================
    % ACCENTLIJN
    % ========================================================

    \noindent
    \begin{tikzpicture}
        \shade[
            left color=fvdHeroPink,
            right color=fvdHeroCyan
        ]
        (0,0) rectangle (25mm,0.8mm);
    \end{tikzpicture}


    % ========================================================
    % ABSTRACT
    % ========================================================

    \if\relax\detokenize{#1}\relax
        % Geen abstract: niets tonen.
    \else

        \vspace{3mm}

        \begin{minipage}{0.76\linewidth}

            {\raggedright
             \sffamily
             \itshape
             \fontsize{9.3}{13.5}\selectfont

    \fi

    \end{tcolorbox}

    % wordt meegenomen.
    \global\let\fvdabstracttext\@empty

% ============================================================
% ABSTRACT BEWAREN
% ============================================================
%
% \xmtitle uit preamble.tex doet intern:
%
%   \begin{abstract}
%       ...
%   \end{abstract}
%   \maketitle
%
% We bewaren hier de inhoud. De chapterstyle uit printstyle.sty
% geeft die daarna door aan \fvdchapterhero.
%
% Het woord "Abstract" wordt hierdoor niet afzonderlijk getoond.
% ============================================================

\@ifundefined{abstract}{%

    \NewDocumentEnvironment{abstract}{+b}{%
        \global\long\def\fvdabstracttext{#1}%
    }{}

}{%

    \RenewDocumentEnvironment{abstract}{+b}{%
        \global\long\def\fvdabstracttext{#1}%
    }{}

}


\makeatother
