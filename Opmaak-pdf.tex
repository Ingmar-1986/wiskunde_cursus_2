% ============================================================
% DEFINITIEVE CHAPTERSTYLE
% ============================================================
%
% Dit is de cruciale koppeling. De oude PDF miste de hero omdat
% gewone \activity-oproepen niet altijd als chapter werden gezet.
% De aangepaste printstyle activeert chapterstyle voor elke
% activity; deze definitie tekent vervolgens altijd de hero.
% ============================================================

\renewcommand\chapterstyle{%
  \def\maketitle{%
    \addtocounter{titlenumber}{1}%

    \phantomsection
    \addcontentsline{toc}{section}{%
      \protect\numberline{\thetitlenumber}\@title
    }%

    \renewcommand\xmsection\subsection
    \renewcommand\xmsubsection\subsubsection
    \renewcommand{\thesubsection}{%
      \thetitlenumber.\arabic{subsection}%
    }
    \renewcommand{\thesubsubsection}{%
      \thesubsection.\arabic{subsubsection}%
    }

    \setcounter{sectiontitlenumber}{0}%
    \setcounter{problem}{0}%
    \setcounter{example}{0}%
    \setcounter{exercise}{0}%
    \setcounter{question}{0}%
    \setcounter{subsection}{0}%
    \setcounter{subsubsection}{0}%

    \@ifundefined{c@proposition}{}{\setcounter{proposition}{0}}%
    \@ifundefined{c@definition}{}{\setcounter{definition}{0}}%
    \@ifundefined{c@explanation}{}{\setcounter{explanation}{0}}%
    \@ifundefined{c@remark}{}{\setcounter{remark}{0}}%
    \@ifundefined{c@notation}{}{\setcounter{notation}{0}}%
    \@ifundefined{c@observation}{}{\setcounter{observation}{0}}%
    \@ifundefined{c@theorem}{}{\setcounter{theorem}{0}}%
    \@ifundefined{c@warning}{}{\setcounter{warning}{0}}%
    \@ifundefined{c@xmuitweiding}{}{\setcounter{xmuitweiding}{0}}%

    \fvdchapterhero{\fvdabstracttext}%

    \markboth
      {\thetitlenumber\enspace\@title}
      {\thetitlenumber\enspace\@title}%

    \thispagestyle{firstpage}%
    \afterpage{\thispagestyle{otherpage}}%
  }%
}

% ============================================================
% SECTIETITELS

% Geen #1 in \titleformat en geen complexe after-code:
% daardoor blijven deze definities compatibel met titlesec 2.17.
% ============================================================

\titleformat{\subsection}[hang]
  {\sffamily\bfseries\Large\color{fvdHeroPinkBright}}
  {%
    \raisebox{-1.5mm}{%
      \begin{tikzpicture}[baseline]
        \fill[fvdHeroPinkBright]
          (0,0) circle[radius=4.2mm];
        \node[
          text=white,
          font=\sffamily\bfseries\small
        ] at (0,0) {\thesubsection};
      \end{tikzpicture}%
    }%
  }
  {3.5mm}
  {}

\titlespacing*{\subsection}
  {0pt}{7mm}{3.5mm}

\titleformat{\subsubsection}[hang]
  {\sffamily\bfseries\large\color{fvdNavyLight}}
  {\thesubsubsection}
  {2.5mm}
  {}

\titlespacing*{\subsubsection}
  {0pt}{4.5mm}{2.5mm}

\titleformat{\paragraph}[hang]
  {\sffamily\bfseries\normalsize\color{fvdPink}}
  {\theparagraph}
  {2mm}
  {}

\titlespacing*{\paragraph}
  {0pt}{4mm}{2mm}
