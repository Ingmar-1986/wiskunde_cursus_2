% ============================================================
% XOURSE OF AFZONDERLIJK HOOFDSTUK
% ============================================================

\newif\iffvdxourse
\newif\iffvdinsideactivity

\fvdinsideactivityfalse

\@ifclassloaded{xourse}{%
    \fvdxoursetrue
}{%
    \fvdxoursefalse

    % Bij het afzonderlijk compileren van een hoofdstuk
    % wordt \maketitle als hoofdstuktitel behandeld.
    \fvdinsideactivitytrue
}


% ============================================================
% THEMA'S EN HOOFDSTUKKEN IN EEN XOURSE
% ============================================================

\iffvdxourse

    \let\fvdoriginalpart\part
    \let\fvdoriginalactivity\activity

    % --------------------------------------------------------
    % THEMA
    % --------------------------------------------------------

    \RenewDocumentCommand{\part}{o m}{%

        \stepcounter{fvdthema}%
        \setcounter{fvdhoofdstuk}{0}%

        \gdef\fvdthematitel{#2}%

        \IfNoValueTF{#1}{%
            \fvdoriginalpart{#2}%
        }{%
            \fvdoriginalpart[#1]{#2}%
        }%
    }

    % --------------------------------------------------------
    % HOOFDSTUK
    % --------------------------------------------------------

    \RenewDocumentCommand{\activity}{o m}{%

        \stepcounter{fvdhoofdstuk}%

        \clearpage
        \global\fvdinsideactivitytrue

        \IfNoValueTF{#1}{%
            \fvdoriginalactivity{#2}%
        }{%
            \fvdoriginalactivity[#1]{#2}%
        }%

        \global\fvdinsideactivityfalse
    }

\fi


% ============================================================
% HANDMATIGE HOOFDSTUKGEGEVENS
% ============================================================
%
% Voor het afzonderlijk compileren van een hoofdstuk:
%
% \fvdchapterdata
%     {1}
%     {Getallen en bewerkingen}
%     {2}
%
% ============================================================

\providecommand{\fvdchapterdata}[3]{%
    \setcounter{fvdthema}{#1}%
    \gdef\fvdthematitel{#2}%
    \setcounter{fvdhoofdstuk}{#3}%
}
