  \noindent

    \node[
      circle,
      fill=fvdHeroPinkBright,
      minimum size=9mm,
      inner sep=0pt,
      text=white,
      font=\sffamily\bfseries\large
    ] (fvdnumber) {\thefvdsection};

    \node[
      anchor=west,
      text=fvdHeroPinkBright,
      font=\sffamily\bfseries\Large
    ] (fvdtitle) at ([xshift=4mm]fvdnumber.east) {#1};
  \end{tikzpicture}%
